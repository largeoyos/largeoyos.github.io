38class EntropyModel(nn.Module):
39    r"""Entropy model base class.
40
41    Args:
42        likelihood_bound (float): minimum likelihood bound
43        entropy_coder (str, optional): set the entropy coder to use, use default
44            one if None
45        entropy_coder_precision (int): set the entropy coder precision
46    """
47
48    def __init__(
49        self, likelihood_bound=1e-9, entropy_coder=None, entropy_coder_precision=16
50    ):
51        super().__init__()
52        self.entropy_coder = None
53        self.entropy_coder_precision = int(entropy_coder_precision)
54
55        self.use_likelihood_bound = likelihood_bound > 0
56        if self.use_likelihood_bound:
57            self.likelihood_lower_bound = LowerBound(likelihood_bound)
58
59        # to be filled on update()
60        self.register_buffer("_offset", torch.IntTensor())
61        self.register_buffer("_quantized_cdf", torch.IntTensor())
62        self.register_buffer("_cdf_length", torch.IntTensor())
63
64    def forward(self, *args):
65        raise NotImplementedError()
66
67    def _check_entropy_coder(self):
68        if self.entropy_coder is None:
69            self.entropy_coder = _EntropyCoder()
70
71    def _quantize(self, inputs, mode, means=None):
72        if mode not in ("dequantize", "symbols"):
73            raise ValueError(f'Invalid quantization mode: "{mode}"')
74
75        outputs = inputs.clone()
76        if means is not None:
77            outputs -= means
78
79        outputs = torch.round(outputs)
80
81        if mode == "dequantize":
82            if means is not None:
83                outputs += means
84            return outputs
85
86        assert mode == "symbols", mode
87        outputs = outputs.int()
88        return outputs
89
90    @staticmethod
91    def _dequantize(inputs, means=None):
92        if means is not None:
93            outputs = inputs.type_as(means)
94            outputs += means
95        else:
96            outputs = inputs.float()
97        return outputs
98
99    def _pmf_to_cdf(self, pmf, tail_mass, pmf_length, max_length):
100        cdf = torch.zeros((len(pmf_length), max_length + 2), dtype=torch.int32)
101        for i, p in enumerate(pmf):
102            prob = torch.cat((p[: pmf_length[i]], tail_mass[i]), dim=0)
103            _cdf = pmf_to_quantized_cdf(prob, self.entropy_coder_precision)
104            cdf[i, : _cdf.size(0)] = _cdf
105        return cdf
106
107    def _check_cdf_size(self):
108        if self._quantized_cdf.numel() == 0:
109            raise ValueError("Uninitialized CDFs. Run update() first")
110
111        if len(self._quantized_cdf.size()) != 2:
112            raise ValueError(f"Invalid CDF size {self._quantized_cdf.size()}")
113
114    def _check_offsets_size(self):
115        if self._offset.numel() == 0:
116            raise ValueError("Uninitialized offsets. Run update() first")
117
118        if len(self._offset.size()) != 1:
119            raise ValueError(f"Invalid offsets size {self._offset.size()}")
120
121    def _check_cdf_length(self):
122        if self._cdf_length.numel() == 0:
123            raise ValueError("Uninitialized CDF lengths. Run update() first")
124
125        if len(self._cdf_length.size()) != 1:
126            raise ValueError(f"Invalid offsets size {self._cdf_length.size()}")
127
128    def compress(self, inputs, indexes, means=None):
129        """
130        Compress input tensors to char strings.
131
132        Args:
133            inputs (torch.Tensor): input tensors
134            indexes (torch.IntTensor): tensors CDF indexes
135            means (torch.Tensor, optional): optional tensor means
136        """
137        symbols = self._quantize(inputs, "symbols", means)
138
139        if len(inputs.size()) != 4:
140            raise ValueError("Invalid `inputs` size. Expected a 4-D tensor.")
141
142        if inputs.size() != indexes.size():
143            raise ValueError("`inputs` and `indexes` should have the same size.")
144
145        self._check_cdf_size()
146        self._check_cdf_length()
147        self._check_offsets_size()
148
149        strings = []
150        self._check_entropy_coder()
151        for i in range(symbols.size(0)):
152            rv = self.entropy_coder.encode_with_indexes(
153                symbols[i].reshape(-1).int().tolist(),
154                indexes[i].reshape(-1).int().tolist(),
155                self._quantized_cdf.tolist(),
156                self._cdf_length.reshape(-1).int().tolist(),
157                self._offset.reshape(-1).int().tolist(),
158            )
159            strings.append(rv)
160        return strings
161
162    def decompress(self, strings, indexes, means=None):
163        """
164        Decompress char strings to tensors.
165
166        Args:
167            strings (str): compressed tensors
168            indexes (torch.IntTensor): tensors CDF indexes
169            means (torch.Tensor, optional): optional tensor means
170        """
171
172        if not isinstance(strings, (tuple, list)):
173            raise ValueError("Invalid `strings` parameter type.")
174
175        if not len(strings) == indexes.size(0):
176            raise ValueError("Invalid strings or indexes parameters")
177
178        if len(indexes.size()) != 4:
179            raise ValueError("Invalid `indexes` size. Expected a 4-D tensor.")
180
181        self._check_cdf_size()
182        self._check_cdf_length()
183        self._check_offsets_size()
184
185        if means is not None:
186            if means.size()[:-2] != indexes.size()[:-2]:
187                raise ValueError("Invalid means or indexes parameters")
188            if means.size() != indexes.size() and (
189                means.size(2) != 1 or means.size(3) != 1
190            ):
191                raise ValueError("Invalid means parameters")
192
193        cdf = self._quantized_cdf
194        outputs = cdf.new(indexes.size())
195        self._check_entropy_coder()
196        for i, s in enumerate(strings):
197            values = self.entropy_coder.decode_with_indexes(
198                s,
199                indexes[i].reshape(-1).int().tolist(),
200                cdf.tolist(),
201                self._cdf_length.reshape(-1).int().tolist(),
202                self._offset.reshape(-1).int().tolist(),
203            )
204            outputs[i] = torch.Tensor(values).reshape(outputs[i].size())
205        outputs = self._dequantize(outputs, means)
206        return outputs

209class EntropyBottleneck(EntropyModel):
210    r"""Entropy bottleneck layer, introduced by J. Ballé, D. Minnen, S. Singh,
211    S. J. Hwang, N. Johnston, in `"Variational image compression with a scale
212    hyperprior" <https://arxiv.org/abs/1802.01436>`_.
213
214    This is a re-implementation of the entropy bottleneck layer in
215    *tensorflow/compression*. See the original paper and the `tensorflow
216    documentation
217    <https://tensorflow.github.io/compression/docs/entropy_bottleneck.html>`__
218    for an introduction.
219    """
220
221    def __init__(
222        self,
223        channels,
224        *args,
225        tail_mass=1e-9,
226        init_scale=10,
227        filters=(3, 3, 3, 3),
228        **kwargs,
229    ):
230        super().__init__(*args, **kwargs)
231
232        self.channels = int(channels)
233        self.filters = tuple(int(f) for f in filters)
234        self.init_scale = float(init_scale)
235        self.tail_mass = float(tail_mass)
236
237        # Create parameters
238        self._biases = nn.ParameterList()
239        self._factors = nn.ParameterList()
240        self._matrices = nn.ParameterList()
241
242        filters = (1,) + self.filters + (1,)
243        scale = self.init_scale ** (1 / (len(self.filters) + 1))
244        channels = self.channels
245
246        for i in range(len(self.filters) + 1):
247            init = np.log(np.expm1(1 / scale / filters[i + 1]))
248            matrix = torch.Tensor(channels, filters[i + 1], filters[i])
249            matrix.data.fill_(init)
250            self._matrices.append(nn.Parameter(matrix))
251
252            bias = torch.Tensor(channels, filters[i + 1], 1)
253            nn.init.uniform_(bias, -0.5, 0.5)
254            self._biases.append(nn.Parameter(bias))
255
256            if i < len(self.filters):
257                factor = torch.Tensor(channels, filters[i + 1], 1)
258                nn.init.zeros_(factor)
259                self._factors.append(nn.Parameter(factor))
260
261        self.quantiles = nn.Parameter(torch.Tensor(channels, 1, 3))
262        init = torch.Tensor([-self.init_scale, 0, self.init_scale])
263        self.quantiles.data = init.repeat(self.quantiles.size(0), 1, 1)
264
265        target = np.log(2 / self.tail_mass - 1)
266        self.register_buffer("target", torch.Tensor([-target, 0, target]))
267
268    def _medians(self):
269        medians = self.quantiles[:, :, 1:2]
270        return medians
271
272    def update(self, force=False):
273        # Check if we need to update the bottleneck parameters, the offsets are
274        # only computed and stored when the conditonal model is update()'d.
275        if self._offset.numel() > 0 and not force:  # pylint: disable=E0203
276            return
277
278        medians = self.quantiles[:, 0, 1]
279
280        minima = medians - self.quantiles[:, 0, 0]
281        minima = torch.ceil(minima).int()
282        minima = torch.clamp(minima, min=0)
283
284        maxima = self.quantiles[:, 0, 2] - medians
285        maxima = torch.ceil(maxima).int()
286        maxima = torch.clamp(maxima, min=0)
287
288        self._offset = -minima
289
290        pmf_start = medians - minima
291        pmf_length = maxima + minima + 1
292
293        max_length = pmf_length.max()
294        device = pmf_start.device
295        samples = torch.arange(max_length, device=device)
296
297        samples = samples[None, :] + pmf_start[:, None, None]
298
299        half = float(0.5)
300
301        lower = self._logits_cumulative(samples - half, stop_gradient=True)
302        upper = self._logits_cumulative(samples + half, stop_gradient=True)
303        sign = -torch.sign(lower + upper)
304        pmf = torch.abs(torch.sigmoid(sign * upper) - torch.sigmoid(sign * lower))
305
306        pmf = pmf[:, 0, :]
307        tail_mass = torch.sigmoid(lower[:, 0, :1]) + torch.sigmoid(-upper[:, 0, -1:])
308
309        quantized_cdf = self._pmf_to_cdf(pmf, tail_mass, pmf_length, max_length)
310        self._quantized_cdf = quantized_cdf
311        self._cdf_length = pmf_length + 2
312
313    def _logits_cumulative(self, inputs, stop_gradient):
314        # TorchScript not yet working (nn.Mmodule indexing not supported)
315        logits = inputs
316        for i in range(len(self.filters) + 1):
317            matrix = self._matrices[i]
318            if stop_gradient:
319                matrix = matrix.detach()
320            logits = torch.matmul(F.softplus(matrix), logits)
321
322            bias = self._biases[i]
323            if stop_gradient:
324                bias = bias.detach()
325            logits += bias
326
327            if i < len(self._factors):
328                factor = self._factors[i]
329                if stop_gradient:
330                    factor = factor.detach()
331                logits += torch.tanh(factor) * torch.tanh(logits)
332        return logits
333
334    @torch.jit.unused
335    def _likelihood(self, inputs):
336        half = float(0.5)
337        v0 = inputs - half
338        v1 = inputs + half
339        lower = self._logits_cumulative(v0, stop_gradient=False)
340        upper = self._logits_cumulative(v1, stop_gradient=False)
341        sign = -torch.sign(lower + upper)
342        sign = sign.detach()
343        likelihood = torch.abs(
344            torch.sigmoid(sign * upper) - torch.sigmoid(sign * lower)
345        )
346        return likelihood
347
348    def forward(self, x):
349        # Convert to (channels, ... , batch) format
350        x = x.permute(1, 2, 3, 0).contiguous()
351        shape = x.size()
352        values = x.reshape(x.size(0), 1, -1)
353
354        # Add noise or quantize
355
356        outputs = self._quantize(
357            values, "dequantize", self._medians()
358        )
359
360        likelihood = self._likelihood(outputs)
361        if self.use_likelihood_bound:
362            likelihood = self.likelihood_lower_bound(likelihood)
363
364        # Convert back to input tensor shape
365        outputs = outputs.reshape(shape)
366        outputs = outputs.permute(3, 0, 1, 2).contiguous()
367
368        likelihood = likelihood.reshape(shape)
369        likelihood = likelihood.permute(3, 0, 1, 2).contiguous()
370
371        return outputs, likelihood
372
373    @staticmethod
374    def _build_indexes(size):
375        N, C, H, W = size
376        indexes = torch.arange(C).view(1, -1, 1, 1)
377        indexes = indexes.int()
378        return indexes.repeat(N, 1, H, W)
379
380    def compress(self, x):
381        indexes = self._build_indexes(x.size())
382        medians = self._medians().detach().view(1, -1, 1, 1)
383        return super().compress(x, indexes, medians)
384
385    def decompress(self, strings, size):
386        output_size = (len(strings), self._quantized_cdf.size(0), size[0], size[1])
387        indexes = self._build_indexes(output_size)
388        medians = self._medians().detach().view(1, -1, 1, 1)
389        return super().decompress(strings, indexes, medians)

392class GaussianConditional(EntropyModel):
393    r"""Gaussian conditional layer, introduced by J. Ballé, D. Minnen, S. Singh,
394    S. J. Hwang, N. Johnston, in `"Variational image compression with a scale
395    hyperprior" <https://arxiv.org/abs/1802.01436>`_.
396
397    This is a re-implementation of the Gaussian conditional layer in
398    *tensorflow/compression*. See the `tensorflow documentation
399    <https://tensorflow.github.io/compression/docs/api_docs/python/tfc/GaussianConditional.html>`__
400    for more information.
401    """
402
403    def __init__(self, scale_table, *args, scale_bound=0.11, tail_mass=1e-9, **kwargs):
404        super().__init__(*args, **kwargs)
405
406        if not isinstance(scale_table, (type(None), list, tuple)):
407            raise ValueError(f'Invalid type for scale_table "{type(scale_table)}"')
408
409        if isinstance(scale_table, (list, tuple)) and len(scale_table) < 1:
410            raise ValueError(f'Invalid scale_table length "{len(scale_table)}"')
411
412        if scale_table and (
413            scale_table != sorted(scale_table) or any(s <= 0 for s in scale_table)
414        ):
415            raise ValueError(f'Invalid scale_table "({scale_table})"')
416
417        self.register_buffer(
418            "scale_table",
419            self._prepare_scale_table(scale_table) if scale_table else torch.Tensor(),
420        )
421
422        self.register_buffer(
423            "scale_bound",
424            torch.Tensor([float(scale_bound)]) if scale_bound is not None else None,
425        )
426
427        self.tail_mass = float(tail_mass)
428        if scale_bound is None and scale_table:
429            self.lower_bound_scale = LowerBound(self.scale_table[0])
430        elif scale_bound > 0:
431            self.lower_bound_scale = LowerBound(scale_bound)
432        else:
433            raise ValueError("Invalid parameters")
434
435    @staticmethod
436    def _prepare_scale_table(scale_table):
437        return torch.Tensor(tuple(float(s) for s in scale_table))
438
439    def _standardized_cumulative(self, inputs):
440        half = float(0.5)
441        const = float(-(2 ** -0.5))
442        # Using the complementary error function maximizes numerical precision.
443        return half * torch.erfc(const * inputs)
444
445    @staticmethod
446    def _standardized_quantile(quantile):
447        return scipy.stats.norm.ppf(quantile)
448
449    def update_scale_table(self, scale_table, force=False):
450        # Check if we need to update the gaussian conditional parameters, the
451        # offsets are only computed and stored when the conditonal model is
452        # updated.
453        if self._offset.numel() > 0 and not force:
454            return
455        self.scale_table = self._prepare_scale_table(scale_table)
456        self.update()
457
458    def update(self):
459        multiplier = -self._standardized_quantile(self.tail_mass / 2)
460        pmf_center = torch.ceil(self.scale_table * multiplier).int()
461        pmf_length = 2 * pmf_center + 1
462        max_length = torch.max(pmf_length).item()
463
464        device = pmf_center.device
465        samples = torch.abs(
466            torch.arange(max_length, device=device).int() - pmf_center[:, None]
467        )
468        samples_scale = self.scale_table.unsqueeze(1)
469        samples = samples.float()
470        samples_scale = samples_scale.float()
471        upper = self._standardized_cumulative((0.5 - samples) / samples_scale)
472        lower = self._standardized_cumulative((-0.5 - samples) / samples_scale)
473        pmf = upper - lower
474
475        tail_mass = 2 * lower[:, :1]
476
477        quantized_cdf = torch.Tensor(len(pmf_length), max_length + 2)
478        quantized_cdf = self._pmf_to_cdf(pmf, tail_mass, pmf_length, max_length)
479        self._quantized_cdf = quantized_cdf
480        self._offset = -pmf_center
481        self._cdf_length = pmf_length + 2
482
483    def _likelihood(self, inputs, scales, means=None):
484        half = float(0.5)
485
486        if means is not None:
487            values = inputs - means
488        else:
489            values = inputs
490
491        scales = self.lower_bound_scale(scales)
492
493        values = torch.abs(values)
494        upper = self._standardized_cumulative((half - values) / scales)
495        lower = self._standardized_cumulative((-half - values) / scales)
496        likelihood = upper - lower
497
498        return likelihood
499
500    def forward(self, inputs, scales, means=None):
501        outputs = self._quantize(
502            inputs, "dequantize", means
503        )
504        likelihood = self._likelihood(outputs, scales, means)
505        if self.use_likelihood_bound:
506            likelihood = self.likelihood_lower_bound(likelihood)
507        return outputs, likelihood
508
509    def build_indexes(self, scales):
510        scales = self.lower_bound_scale(scales)
511        indexes = scales.new_full(scales.size(), len(self.scale_table) - 1).int()
512        for s in self.scale_table[:-1]:
513            indexes -= (scales <= s).int()
514        return indexes
