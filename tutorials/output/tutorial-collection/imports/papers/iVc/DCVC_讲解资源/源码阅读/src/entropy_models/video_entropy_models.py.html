10class EntropyCoder(object):
11    def __init__(self, entropy_coder_precision=16):
12        super().__init__()
13
14        from .MLCodec_rans import RansEncoder, RansDecoder
15        self.encoder = RansEncoder()
16        self.decoder = RansDecoder()
17        self.entropy_coder_precision = int(entropy_coder_precision)
18        self._offset = None
19        self._quantized_cdf = None
20        self._cdf_length = None
21
22    def encode_with_indexes(self, *args, **kwargs):
23        return self.encoder.encode_with_indexes(*args, **kwargs)
24
25    def decode_with_indexes(self, *args, **kwargs):
26        return self.decoder.decode_with_indexes(*args, **kwargs)
27
28    def set_cdf_states(self, offset, quantized_cdf, cdf_length):
29        self._offset = offset
30        self._quantized_cdf = quantized_cdf
31        self._cdf_length = cdf_length
32
33    @staticmethod
34    def pmf_to_quantized_cdf(pmf, precision=16):
35        from .MLCodec_CXX import pmf_to_quantized_cdf as _pmf_to_quantized_cdf
36        cdf = _pmf_to_quantized_cdf(pmf.tolist(), precision)
37        cdf = torch.IntTensor(cdf)
38        return cdf
39
40    def pmf_to_cdf(self, pmf, tail_mass, pmf_length, max_length):
41        cdf = torch.zeros((len(pmf_length), max_length + 2), dtype=torch.int32)
42        for i, p in enumerate(pmf):
43            prob = torch.cat((p[: pmf_length[i]], tail_mass[i]), dim=0)
44            _cdf = self.pmf_to_quantized_cdf(prob, self.entropy_coder_precision)
45            cdf[i, : _cdf.size(0)] = _cdf
46        return cdf
47
48    def _check_cdf_size(self):
49        if self._quantized_cdf.numel() == 0:
50            raise ValueError("Uninitialized CDFs. Run update() first")
51
52        if len(self._quantized_cdf.size()) != 2:
53            raise ValueError(f"Invalid CDF size {self._quantized_cdf.size()}")
54
55    def _check_offsets_size(self):
56        if self._offset.numel() == 0:
57            raise ValueError("Uninitialized offsets. Run update() first")
58
59        if len(self._offset.size()) != 1:
60            raise ValueError(f"Invalid offsets size {self._offset.size()}")
61
62    def _check_cdf_length(self):
63        if self._cdf_length.numel() == 0:
64            raise ValueError("Uninitialized CDF lengths. Run update() first")
65
66        if len(self._cdf_length.size()) != 1:
67            raise ValueError(f"Invalid offsets size {self._cdf_length.size()}")
68
69    def compress(self, inputs, indexes):
70        """
71        """
72        if len(inputs.size()) != 4:
73            raise ValueError("Invalid `inputs` size. Expected a 4-D tensor.")
74
75        if inputs.size() != indexes.size():
76            raise ValueError("`inputs` and `indexes` should have the same size.")
77        symbols = inputs.int()
78
79        self._check_cdf_size()
80        self._check_cdf_length()
81        self._check_offsets_size()
82
83        assert symbols.size(0) == 1
84        rv = self.encode_with_indexes(
85            symbols[0].reshape(-1).int().tolist(),
86            indexes[0].reshape(-1).int().tolist(),
87            self._quantized_cdf.tolist(),
88            self._cdf_length.reshape(-1).int().tolist(),
89            self._offset.reshape(-1).int().tolist(),
90        )
91        return rv
92
93    def decompress(self, strings, indexes):
94        """
95        Decompress char strings to tensors.
96
97        Args:
98            strings (str): compressed tensors
99            indexes (torch.IntTensor): tensors CDF indexes
100        """
101
102        assert indexes.size(0) == 1
103
104        if len(indexes.size()) != 4:
105            raise ValueError("Invalid `indexes` size. Expected a 4-D tensor.")
106
107        self._check_cdf_size()
108        self._check_cdf_length()
109        self._check_offsets_size()
110
111        cdf = self._quantized_cdf
112        outputs = cdf.new(indexes.size())
113
114        values = self.decode_with_indexes(
115            strings,
116            indexes[0].reshape(-1).int().tolist(),
117            self._quantized_cdf.tolist(),
118            self._cdf_length.reshape(-1).int().tolist(),
119            self._offset.reshape(-1).int().tolist(),
120        )
121        outputs[0] = torch.Tensor(values).reshape(outputs[0].size())
122        return outputs.float()
123
124    def set_stream(self, stream):
125        self.decoder.set_stream(stream)
126
127    def decode_stream(self, indexes):
128        rv = self.decoder.decode_stream(
129            indexes.squeeze().int().tolist(),
130            self._quantized_cdf.tolist(),
131            self._cdf_length.reshape(-1).int().tolist(),
132            self._offset.reshape(-1).int().tolist(),
133        )
134        rv = torch.Tensor(rv).reshape(1, -1, 1, 1)
135        return rv

138class Bitparm(nn.Module):
139    def __init__(self, channel, final=False):
140        super(Bitparm, self).__init__()
141        self.final = final
142        self.h = nn.Parameter(torch.nn.init.normal_(
143            torch.empty(channel).view(1, -1, 1, 1), 0, 0.01))
144        self.b = nn.Parameter(torch.nn.init.normal_(
145            torch.empty(channel).view(1, -1, 1, 1), 0, 0.01))
146        if not final:
147            self.a = nn.Parameter(torch.nn.init.normal_(
148                torch.empty(channel).view(1, -1, 1, 1), 0, 0.01))
149        else:
150            self.a = None
151
152    def forward(self, x):
153        if self.final:
154            return torch.sigmoid(x * F.softplus(self.h) + self.b)
155        else:
156            x = x * F.softplus(self.h) + self.b
157            return x + torch.tanh(x) * torch.tanh(self.a)

160class BitEstimator(nn.Module):
161    def __init__(self, channel):
162        super(BitEstimator, self).__init__()
163        self.f1 = Bitparm(channel)
164        self.f2 = Bitparm(channel)
165        self.f3 = Bitparm(channel)
166        self.f4 = Bitparm(channel, True)
167        self.channel = channel
168        self.entropy_coder = None
169
170    def forward(self, x):
171        x = self.f1(x)
172        x = self.f2(x)
173        x = self.f3(x)
174        return self.f4(x)
175
176    def update(self, force=False):
177        # Check if we need to update the bottleneck parameters, the offsets are
178        # only computed and stored when the conditonal model is update()'d.
179        if self.entropy_coder is not None and not force:  # pylint: disable=E0203
180            return
181
182        self.entropy_coder = EntropyCoder()
183        with torch.no_grad():
184            device = next(self.parameters()).device
185            medians = torch.zeros((self.channel), device=device)
186
187            minima = medians + 50
188            for i in range(50, 1, -1):
189                samples = torch.zeros_like(medians) - i
190                samples = samples[None, :, None, None]
191                probs = self.forward(samples)
192                probs = torch.squeeze(probs)
193                minima = torch.where(probs < torch.zeros_like(medians) + 0.0001,
194                                     torch.zeros_like(medians) + i, minima)
195
196            maxima = medians + 50
197            for i in range(50, 1, -1):
198                samples = torch.zeros_like(medians) + i
199                samples = samples[None, :, None, None]
200                probs = self.forward(samples)
201                probs = torch.squeeze(probs)
202                maxima = torch.where(probs > torch.zeros_like(medians) + 0.9999,
203                                     torch.zeros_like(medians) + i, maxima)
204
205            minima = minima.int()
206            maxima = maxima.int()
207
208            offset = -minima
209
210            pmf_start = medians - minima
211            pmf_length = maxima + minima + 1
212
213            max_length = pmf_length.max()
214            device = pmf_start.device
215            samples = torch.arange(max_length, device=device)
216
217            samples = samples[None, :] + pmf_start[:, None, None]
218
219            half = float(0.5)
220
221            lower = self.forward(samples - half).squeeze(0)
222            upper = self.forward(samples + half).squeeze(0)
223            pmf = upper - lower
224
225            pmf = pmf[:, 0, :]
226            tail_mass = lower[:, 0, :1] + (1.0 - upper[:, 0, -1:])
227
228            quantized_cdf = self.entropy_coder.pmf_to_cdf(pmf, tail_mass, pmf_length, max_length)
229            cdf_length = pmf_length + 2
230            self.entropy_coder.set_cdf_states(offset, quantized_cdf, cdf_length)
231
232    @staticmethod
233    def build_indexes(size):
234        N, C, H, W = size
235        indexes = torch.arange(C).view(1, -1, 1, 1)
236        indexes = indexes.int()
237        return indexes.repeat(N, 1, H, W)
238
239    def compress(self, x):
240        indexes = self.build_indexes(x.size())
241        return self.entropy_coder.compress(x, indexes)
242
243    def decompress(self, strings, size):
244        output_size = (1, self.entropy_coder._quantized_cdf.size(0), size[0], size[1])
245        indexes = self.build_indexes(output_size)
246        return self.entropy_coder.decompress(strings, indexes)

249class GaussianEncoder(object):
250    def __init__(self):
251        self.scale_table = self.get_scale_table()
252        self.entropy_coder = None
253
254    @staticmethod
255    def get_scale_table(min=0.01, max=16, levels=64):  # pylint: disable=W0622
256        return torch.exp(torch.linspace(math.log(min), math.log(max), levels))
257
258    def update(self, force=False):
259        if self.entropy_coder is not None and not force:
260            return
261        self.entropy_coder = EntropyCoder()
262
263        pmf_center = torch.zeros_like(self.scale_table) + 50
264        scales = torch.zeros_like(pmf_center) + self.scale_table
265        mu = torch.zeros_like(scales)
266        gaussian = torch.distributions.laplace.Laplace(mu, scales)
267        for i in range(50, 1, -1):
268            samples = torch.zeros_like(pmf_center) + i
269            probs = gaussian.cdf(samples)
270            probs = torch.squeeze(probs)
271            pmf_center = torch.where(probs > torch.zeros_like(pmf_center) + 0.9999,
272                                     torch.zeros_like(pmf_center) + i, pmf_center)
273
274        pmf_center = pmf_center.int()
275        pmf_length = 2 * pmf_center + 1
276        max_length = torch.max(pmf_length).item()
277
278        device = pmf_center.device
279        samples = torch.arange(max_length, device=device) - pmf_center[:, None]
280        samples = samples.float()
281
282        scales = torch.zeros_like(samples) + self.scale_table[:, None]
283        mu = torch.zeros_like(scales)
284        gaussian = torch.distributions.laplace.Laplace(mu, scales)
285
286        upper = gaussian.cdf(samples + 0.5)
287        lower = gaussian.cdf(samples - 0.5)
288        pmf = upper - lower
289
290        tail_mass = 2 * lower[:, :1]
291
292        quantized_cdf = torch.Tensor(len(pmf_length), max_length + 2)
293        quantized_cdf = self.entropy_coder.pmf_to_cdf(pmf, tail_mass, pmf_length, max_length)
294        self.entropy_coder.set_cdf_states(-pmf_center, quantized_cdf, pmf_length+2)
295
296    def build_indexes(self, scales):
297        scales = torch.maximum(scales, torch.zeros_like(scales) + 1e-5)
298        indexes = scales.new_full(scales.size(), len(self.scale_table) - 1).int()
299        for s in self.scale_table[:-1]:
300            indexes -= (scales <= s).int()
301        return indexes
302
303    def compress(self, x, scales):
304        indexes = self.build_indexes(scales)
305        return self.entropy_coder.compress(x, indexes)
306
307    def decompress(self, strings, scales):
308        indexes = self.build_indexes(scales)
309        return self.entropy_coder.decompress(strings, indexes)
310
311    def set_stream(self, stream):
312        self.entropy_coder.set_stream(stream)
313
314    def decode_stream(self, scales):
315        indexes = self.build_indexes(scales)
316        return self.entropy_coder.decode_stream(indexes)
