15class DCVC_net(nn.Module):
16    def __init__(self):
17        super().__init__()
18        out_channel_mv = 128
19        out_channel_N = 64
20        out_channel_M = 96
21
22        self.out_channel_mv = out_channel_mv
23        self.out_channel_N = out_channel_N
24        self.out_channel_M = out_channel_M
25
26        self.bitEstimator_z = BitEstimator(out_channel_N)
27        self.bitEstimator_z_mv = BitEstimator(out_channel_N)
28
29        self.feature_extract = nn.Sequential(
30            nn.Conv2d(3, out_channel_N, 3, stride=1, padding=1),
31            ResBlock(out_channel_N, out_channel_N, 3),
32        )
33
34        self.context_refine = nn.Sequential(
35            ResBlock(out_channel_N, out_channel_N, 3),
36            nn.Conv2d(out_channel_N, out_channel_N, 3, stride=1, padding=1),
37        )
38
39        self.gaussian_encoder = GaussianEncoder()
40
41        self.mvEncoder = nn.Sequential(
42            nn.Conv2d(2, out_channel_mv, 3, stride=2, padding=1),
43            GDN(out_channel_mv),
44            nn.Conv2d(out_channel_mv, out_channel_mv, 3, stride=2, padding=1),
45            GDN(out_channel_mv),
46            nn.Conv2d(out_channel_mv, out_channel_mv, 3, stride=2, padding=1),
47            GDN(out_channel_mv),
48            nn.Conv2d(out_channel_mv, out_channel_mv, 3, stride=2, padding=1),
49        )
50
51        self.mvDecoder_part1 = nn.Sequential(
52            nn.ConvTranspose2d(out_channel_mv, out_channel_mv, 3,
53                               stride=2, padding=1, output_padding=1),
54            GDN(out_channel_mv, inverse=True),
55            nn.ConvTranspose2d(out_channel_mv, out_channel_mv, 3,
56                               stride=2, padding=1, output_padding=1),
57            GDN(out_channel_mv, inverse=True),
58            nn.ConvTranspose2d(out_channel_mv, out_channel_mv, 3,
59                               stride=2, padding=1, output_padding=1),
60            GDN(out_channel_mv, inverse=True),
61            nn.ConvTranspose2d(out_channel_mv, 2, 3, stride=2, padding=1, output_padding=1),
62        )
63
64        self.mvDecoder_part2 = nn.Sequential(
65            nn.Conv2d(5, 64, 3, stride=1, padding=1),
66            nn.LeakyReLU(negative_slope=0.1),
67            nn.Conv2d(64, 64, 3, stride=1, padding=1),
68            nn.LeakyReLU(negative_slope=0.1),
69            nn.Conv2d(64, 64, 3, stride=1, padding=1),
70            nn.LeakyReLU(negative_slope=0.1),
71            nn.Conv2d(64, 64, 3, stride=1, padding=1),
72            nn.LeakyReLU(negative_slope=0.1),
73            nn.Conv2d(64, 64, 3, stride=1, padding=1),
74            nn.LeakyReLU(negative_slope=0.1),
75            nn.Conv2d(64, 64, 3, stride=1, padding=1),
76            nn.LeakyReLU(negative_slope=0.1),
77            nn.Conv2d(64, 2, 3, stride=1, padding=1),
78        )
79
80        self.contextualEncoder = nn.Sequential(
81            nn.Conv2d(out_channel_N+3, out_channel_N, 5, stride=2, padding=2),
82            GDN(out_channel_N),
83            ResBlock_LeakyReLU_0_Point_1(out_channel_N),
84            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
85            GDN(out_channel_N),
86            ResBlock_LeakyReLU_0_Point_1(out_channel_N),
87            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
88            GDN(out_channel_N),
89            nn.Conv2d(out_channel_N, out_channel_M, 5, stride=2, padding=2),
90        )
91
92        self.contextualDecoder_part1 = nn.Sequential(
93            subpel_conv3x3(out_channel_M, out_channel_N, 2),
94            GDN(out_channel_N, inverse=True),
95            subpel_conv3x3(out_channel_N, out_channel_N, 2),
96            GDN(out_channel_N, inverse=True),
97            ResBlock_LeakyReLU_0_Point_1(out_channel_N),
98            subpel_conv3x3(out_channel_N, out_channel_N, 2),
99            GDN(out_channel_N, inverse=True),
100            ResBlock_LeakyReLU_0_Point_1(out_channel_N),
101            subpel_conv3x3(out_channel_N, out_channel_N, 2),
102        )
103
104        self.contextualDecoder_part2 = nn.Sequential(
105            nn.Conv2d(out_channel_N*2, out_channel_N, 3, stride=1, padding=1),
106            ResBlock(out_channel_N, out_channel_N, 3),
107            ResBlock(out_channel_N, out_channel_N, 3),
108            nn.Conv2d(out_channel_N, 3, 3, stride=1, padding=1),
109        )
110
111        self.priorEncoder = nn.Sequential(
112            nn.Conv2d(out_channel_M, out_channel_N, 3, stride=1, padding=1),
113            nn.LeakyReLU(inplace=True),
114            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
115            nn.LeakyReLU(inplace=True),
116            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
117        )
118
119        self.priorDecoder = nn.Sequential(
120            nn.ConvTranspose2d(out_channel_N, out_channel_M, 5,
121                               stride=2, padding=2, output_padding=1),
122            nn.LeakyReLU(inplace=True),
123            nn.ConvTranspose2d(out_channel_M, out_channel_M, 5,
124                               stride=2, padding=2, output_padding=1),
125            nn.LeakyReLU(inplace=True),
126            nn.ConvTranspose2d(out_channel_M, out_channel_M, 3, stride=1, padding=1)
127        )
128
129        self.mvpriorEncoder = nn.Sequential(
130            nn.Conv2d(out_channel_mv, out_channel_N, 3, stride=1, padding=1),
131            nn.LeakyReLU(inplace=True),
132            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
133            nn.LeakyReLU(inplace=True),
134            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
135        )
136
137        self.mvpriorDecoder = nn.Sequential(
138            nn.ConvTranspose2d(out_channel_N, out_channel_N, 5,
139                               stride=2, padding=2, output_padding=1),
140            nn.LeakyReLU(inplace=True),
141            nn.ConvTranspose2d(out_channel_N, out_channel_N * 3 // 2, 5,
142                               stride=2, padding=2, output_padding=1),
143            nn.LeakyReLU(inplace=True),
144            nn.ConvTranspose2d(out_channel_N * 3 // 2, out_channel_mv*2, 3, stride=1, padding=1)
145        )
146
147        self.entropy_parameters = nn.Sequential(
148            nn.Conv2d(out_channel_M * 12 // 3, out_channel_M * 10 // 3, 1),
149            nn.LeakyReLU(inplace=True),
150            nn.Conv2d(out_channel_M * 10 // 3, out_channel_M * 8 // 3, 1),
151            nn.LeakyReLU(inplace=True),
152            nn.Conv2d(out_channel_M * 8 // 3, out_channel_M * 6 // 3, 1),
153        )
154
155        self.auto_regressive = MaskedConv2d(
156            out_channel_M, 2 * out_channel_M, kernel_size=5, padding=2, stride=1
157        )
158
159        self.auto_regressive_mv = MaskedConv2d(
160            out_channel_mv, 2 * out_channel_mv, kernel_size=5, padding=2, stride=1
161        )
162
163        self.entropy_parameters_mv = nn.Sequential(
164            nn.Conv2d(out_channel_mv * 12 // 3, out_channel_mv * 10 // 3, 1),
165            nn.LeakyReLU(inplace=True),
166            nn.Conv2d(out_channel_mv * 10 // 3, out_channel_mv * 8 // 3, 1),
167            nn.LeakyReLU(inplace=True),
168            nn.Conv2d(out_channel_mv * 8 // 3, out_channel_mv * 6 // 3, 1),
169        )
170
171        self.temporalPriorEncoder = nn.Sequential(
172            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
173            GDN(out_channel_N),
174            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
175            GDN(out_channel_N),
176            nn.Conv2d(out_channel_N, out_channel_N, 5, stride=2, padding=2),
177            GDN(out_channel_N),
178            nn.Conv2d(out_channel_N, out_channel_M, 5, stride=2, padding=2),
179        )
180
181        self.opticFlow = ME_Spynet()

183    def motioncompensation(self, ref, mv):
184        ref_feature = self.feature_extract(ref)
185        prediction_init = flow_warp(ref_feature, mv)
186        context = self.context_refine(prediction_init)
187
188        return context

190    def mv_refine(self, ref, mv):
191        return self.mvDecoder_part2(torch.cat((mv, ref), 1)) + mv

193    def quantize(self, inputs, mode, means=None):
194        assert(mode == "dequantize")
195        outputs = inputs.clone()
196        outputs -= means
197        outputs = torch.round(outputs)
198        outputs += means
199        return outputs
200
201    def feature_probs_based_sigma(self, feature, mean, sigma):
202        outputs = self.quantize(
203            feature, "dequantize", mean
204        )
205        values = outputs - mean
206        mu = torch.zeros_like(sigma)
207        sigma = sigma.clamp(1e-5, 1e10)
208        gaussian = torch.distributions.laplace.Laplace(mu, sigma)
209        probs = gaussian.cdf(values + 0.5) - gaussian.cdf(values - 0.5)
210        total_bits = torch.sum(torch.clamp(-1.0 * torch.log(probs + 1e-5) / math.log(2.0), 0, 50))
211        return total_bits, probs
212
213    def iclr18_estrate_bits_z(self, z):
214        prob = self.bitEstimator_z(z + 0.5) - self.bitEstimator_z(z - 0.5)
215        total_bits = torch.sum(torch.clamp(-1.0 * torch.log(prob + 1e-5) / math.log(2.0), 0, 50))
216        return total_bits, prob
217
218    def iclr18_estrate_bits_z_mv(self, z_mv):
219        prob = self.bitEstimator_z_mv(z_mv + 0.5) - self.bitEstimator_z_mv(z_mv - 0.5)
220        total_bits = torch.sum(torch.clamp(-1.0 * torch.log(prob + 1e-5) / math.log(2.0), 0, 50))
221        return total_bits, prob

228    def encode_decode(self, ref_frame, input_image, output_path):
229        encoded = self.encode(ref_frame, input_image, output_path)
230        decoded = self.decode(ref_frame, output_path)
231        encoded['recon_image'] = decoded
232        return encoded

234    def encode(self, ref_frame, input_image, output_path):
235        from ..utils.stream_helper import encode_p
236        N, C, H, W = ref_frame.size()
237        compressed = self.compress(ref_frame, input_image)
238        mv_y_string = compressed['mv_y_string']
239        mv_z_string = compressed['mv_z_string']
240        y_string = compressed['y_string']
241        z_string = compressed['z_string']
242        encode_p(H, W, mv_y_string, mv_z_string, y_string, z_string, output_path)
243        return {
244            'bpp_mv_y': compressed['bpp_mv_y'],
245            'bpp_mv_z': compressed['bpp_mv_z'],
246            'bpp_y': compressed['bpp_y'],
247            'bpp_z': compressed['bpp_z'],
248            'bpp': compressed['bpp'],

251    def decode(self, ref_frame, input_path):
252        from ..utils.stream_helper import decode_p
253        height, width, mv_y_string, mv_z_string, y_string, z_string = decode_p(input_path)

257    def compress_ar(self, y, kernel_size, context_prediction, params, entropy_parameters):
258        kernel_size = 5
259        padding = (kernel_size - 1) // 2
260
261        height = y.size(2)
262        width = y.size(3)
263
264        y_hat = F.pad(y, (padding, padding, padding, padding))
265        y_q = torch.zeros_like(y)
266        y_scales = torch.zeros_like(y)
267
268        for h in range(height):
269            for w in range(width):
270                y_crop = y_hat[0:1, :, h:h + kernel_size, w:w + kernel_size]
271                ctx_p = F.conv2d(
272                    y_crop,
273                    context_prediction.weight,
274                    bias=context_prediction.bias,
275                )
276
277                p = params[0:1, :, h:h + 1, w:w + 1]
278                gaussian_params = entropy_parameters(torch.cat((p, ctx_p), dim=1))
279                means_hat, scales_hat = gaussian_params.chunk(2, 1)
280
281                y_crop = y_crop[0:1, :, padding:padding+1, padding:padding+1]
282                y_crop_q = torch.round(y_crop - means_hat)
283                y_hat[0, :, h + padding, w + padding] = (y_crop_q + means_hat)[0, :, 0, 0]
284                y_q[0, :, h, w] = y_crop_q[0, :, 0, 0]
285                y_scales[0, :, h, w] = scales_hat[0, :, 0, 0]
286        # change to channel last
287        y_q = y_q.permute(0, 2, 3, 1)
288        y_scales = y_scales.permute(0, 2, 3, 1)
289        y_string = self.gaussian_encoder.compress(y_q, y_scales)
290        y_hat = y_hat[:, :, padding:-padding, padding:-padding]
291        return y_string, y_hat

293    def decompress_ar(self, y_string, channel, height, width, downsample, kernel_size,
294                      context_prediction, params, entropy_parameters):
295        device = next(self.parameters()).device
296        padding = (kernel_size - 1) // 2
297
298        y_size = get_downsampled_shape(height, width, downsample)
299        y_height = y_size[0]
300        y_width = y_size[1]
301
302        y_hat = torch.zeros(
303            (1, channel, y_height + 2 * padding, y_width + 2 * padding),
304            device=params.device,
305        )
306
307        self.gaussian_encoder.set_stream(y_string)
308
309        for h in range(y_height):
310            for w in range(y_width):
311                # only perform the 5x5 convolution on a cropped tensor
312                # centered in (h, w)
313                y_crop = y_hat[0:1, :, h:h + kernel_size, w:w + kernel_size]
314                ctx_p = F.conv2d(
315                    y_crop,
316                    context_prediction.weight,
317                    bias=context_prediction.bias,
318                )
319                p = params[0:1, :, h:h + 1, w:w + 1]
320                gaussian_params = entropy_parameters(torch.cat((p, ctx_p), dim=1))
321                means_hat, scales_hat = gaussian_params.chunk(2, 1)
322                rv = self.gaussian_encoder.decode_stream(scales_hat)
323                rv = rv.to(device)
324                rv = rv + means_hat
325                y_hat[0, :, h + padding: h + padding + 1, w + padding: w + padding + 1] = rv
326
327        y_hat = y_hat[:, :, padding:-padding, padding:-padding]
328        return y_hat

330    def compress(self, referframe, input_image):
331        device = input_image.device
332        estmv = self.opticFlow(input_image, referframe)
333        mvfeature = self.mvEncoder(estmv)
334        z_mv = self.mvpriorEncoder(mvfeature)
335        compressed_z_mv = torch.round(z_mv)
336        mv_z_string = self.bitEstimator_z_mv.compress(compressed_z_mv)
337        mv_z_size = [compressed_z_mv.size(2), compressed_z_mv.size(3)]
338        mv_z_hat = self.bitEstimator_z_mv.decompress(mv_z_string, mv_z_size)
339        mv_z_hat = mv_z_hat.to(device)
340
341        params_mv = self.mvpriorDecoder(mv_z_hat)
342        mv_y_string, mv_y_hat = self.compress_ar(mvfeature, 5, self.auto_regressive_mv,
343                                                 params_mv, self.entropy_parameters_mv)
344
345        quant_mv_upsample = self.mvDecoder_part1(mv_y_hat)
346        quant_mv_upsample_refine = self.mv_refine(referframe, quant_mv_upsample)
347        context = self.motioncompensation(referframe, quant_mv_upsample_refine)
348
349        temporal_prior_params = self.temporalPriorEncoder(context)
350        feature = self.contextualEncoder(torch.cat((input_image, context), dim=1))
351        z = self.priorEncoder(feature)
352        compressed_z = torch.round(z)
353        z_string = self.bitEstimator_z.compress(compressed_z)
354        z_size = [compressed_z.size(2), compressed_z.size(3)]
355        z_hat = self.bitEstimator_z.decompress(z_string, z_size)
356        z_hat = z_hat.to(device)
357
358        params = self.priorDecoder(z_hat)
359        y_string, y_hat = self.compress_ar(feature, 5, self.auto_regressive,
360                                           torch.cat((temporal_prior_params, params), dim=1), self.entropy_parameters)
361
362        recon_image_feature = self.contextualDecoder_part1(y_hat)
363        recon_image = self.contextualDecoder_part2(torch.cat((recon_image_feature, context), dim=1))
364
365        im_shape = input_image.size()
366        pixel_num = im_shape[0] * im_shape[2] * im_shape[3]
367        bpp_y = len(y_string) * 8 / pixel_num
368        bpp_z = len(z_string) * 8 / pixel_num
369        bpp_mv_y = len(mv_y_string) * 8 / pixel_num
370        bpp_mv_z = len(mv_z_string) * 8 / pixel_num
371
372        bpp = bpp_y + bpp_z + bpp_mv_y + bpp_mv_z
373
374        return {"bpp_mv_y": bpp_mv_y,
375                "bpp_mv_z": bpp_mv_z,
376                "bpp_y": bpp_y,
377                "bpp_z": bpp_z,
378                "bpp": bpp,
379                "recon_image": recon_image,
380                "mv_y_string": mv_y_string,
381                "mv_z_string": mv_z_string,
382                "y_string": y_string,
383                "z_string": z_string,
384                }
385
386    def decompress(self, referframe, mv_y_string, mv_z_string, y_string, z_string, height, width):
387        device = next(self.parameters()).device
388        mv_z_size = get_downsampled_shape(height, width, 64)
389        mv_z_hat = self.bitEstimator_z_mv.decompress(mv_z_string, mv_z_size)
390        mv_z_hat = mv_z_hat.to(device)
391        params_mv = self.mvpriorDecoder(mv_z_hat)
392        mv_y_hat = self.decompress_ar(mv_y_string, self.out_channel_mv, height, width, 16, 5,
393                                      self.auto_regressive_mv, params_mv,
394                                      self.entropy_parameters_mv)
395
396        quant_mv_upsample = self.mvDecoder_part1(mv_y_hat)
397        quant_mv_upsample_refine = self.mv_refine(referframe, quant_mv_upsample)
398        context = self.motioncompensation(referframe, quant_mv_upsample_refine)
399        temporal_prior_params = self.temporalPriorEncoder(context)
400
401        z_size = get_downsampled_shape(height, width, 64)
402        z_hat = self.bitEstimator_z.decompress(z_string, z_size)
403        z_hat = z_hat.to(device)
404        params = self.priorDecoder(z_hat)
405        y_hat = self.decompress_ar(y_string, self.out_channel_M, height, width, 16, 5,
406                                   self.auto_regressive, torch.cat((temporal_prior_params, params), dim=1),
407                                   self.entropy_parameters)
408        recon_image_feature = self.contextualDecoder_part1(y_hat)
409        recon_image = self.contextualDecoder_part2(torch.cat((recon_image_feature, context), dim=1))
410        recon_image = recon_image.clamp(0, 1)
411
412        return recon_image

414    def forward(self, referframe, input_image):
415        estmv = self.opticFlow(input_image, referframe)
416        mvfeature = self.mvEncoder(estmv)
417        z_mv = self.mvpriorEncoder(mvfeature)
418        compressed_z_mv = torch.round(z_mv)
419        params_mv = self.mvpriorDecoder(compressed_z_mv)
420
421        quant_mv = torch.round(mvfeature)
422
423        ctx_params_mv = self.auto_regressive_mv(quant_mv)
424        gaussian_params_mv = self.entropy_parameters_mv(
425            torch.cat((params_mv, ctx_params_mv), dim=1)
426        )
427        means_hat_mv, scales_hat_mv = gaussian_params_mv.chunk(2, 1)
428
429        quant_mv_upsample = self.mvDecoder_part1(quant_mv)
430
431        quant_mv_upsample_refine = self.mv_refine(referframe, quant_mv_upsample)
432
433        context = self.motioncompensation(referframe, quant_mv_upsample_refine)
434
435        temporal_prior_params = self.temporalPriorEncoder(context)
436
437        feature = self.contextualEncoder(torch.cat((input_image, context), dim=1))
438        z = self.priorEncoder(feature)
439        compressed_z = torch.round(z)
440        params = self.priorDecoder(compressed_z)
441
442        feature_renorm = feature
443
444        compressed_y_renorm = torch.round(feature_renorm)
445
446        ctx_params = self.auto_regressive(compressed_y_renorm)
447        gaussian_params = self.entropy_parameters(
448            torch.cat((temporal_prior_params, params, ctx_params), dim=1)
449        )
450        means_hat, scales_hat = gaussian_params.chunk(2, 1)
451
452        recon_image_feature = self.contextualDecoder_part1(compressed_y_renorm)
453        recon_image = self.contextualDecoder_part2(torch.cat((recon_image_feature, context), dim=1))
454
455        total_bits_y, _ = self.feature_probs_based_sigma(
456            feature_renorm, means_hat, scales_hat)
457        total_bits_mv, _ = self.feature_probs_based_sigma(mvfeature, means_hat_mv, scales_hat_mv)
458        total_bits_z, _ = self.iclr18_estrate_bits_z(compressed_z)
459        total_bits_z_mv, _ = self.iclr18_estrate_bits_z_mv(compressed_z_mv)
460
461        im_shape = input_image.size()
462        pixel_num = im_shape[0] * im_shape[2] * im_shape[3]
463        bpp_y = total_bits_y / pixel_num
464        bpp_z = total_bits_z / pixel_num
465        bpp_mv_y = total_bits_mv / pixel_num
466        bpp_mv_z = total_bits_z_mv / pixel_num
467
468        bpp = bpp_y + bpp_z + bpp_mv_y + bpp_mv_z
469
470        return {"bpp_mv_y": bpp_mv_y,
471                "bpp_mv_z": bpp_mv_z,
472                "bpp_y": bpp_y,
473                "bpp_z": bpp_z,
474                "bpp": bpp,
475                "recon_image": recon_image,
476                "context": context,
477                }
